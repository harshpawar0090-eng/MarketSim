cmake_minimum_required(VERSION 3.16)
project(MarketSim LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(MARKETSIM_BUILD_CONSOLE "Build the console application (marketsim)" ON)
option(MARKETSIM_BUILD_GUI "Build the Qt 6 GUI (marketsim_gui) if Qt 6 is found" ON)

if(MSVC)
    set(MARKETSIM_WARNINGS /W4)
else()
    set(MARKETSIM_WARNINGS -Wall -Wextra)
endif()

# ---------------------------------------------------------------------------
# Backend core: the existing modules, unchanged, built once and shared by the
# console app and the GUI. (Add Historical / Backtest here when the GUI needs
# them in Phase 5B.3.)
# ---------------------------------------------------------------------------
set(MARKETSIM_CORE_DIRS
    Market
    Trading
    Portfolio
    Simulation
    Execution
    Broker
    Risk
    Analytics
    Historical
    Backtest
    App
)
set(MARKETSIM_CORE_SOURCES "")
foreach(dir IN LISTS MARKETSIM_CORE_DIRS)
    file(GLOB dir_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${dir}/*.cpp")
    list(APPEND MARKETSIM_CORE_SOURCES ${dir_sources})
endforeach()

add_library(marketsim_core STATIC ${MARKETSIM_CORE_SOURCES})
target_include_directories(marketsim_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(marketsim_core PRIVATE ${MARKETSIM_WARNINGS})

# ---------------------------------------------------------------------------
# Console application (unchanged sources: main.cpp + UI/*.cpp)
# ---------------------------------------------------------------------------
if(MARKETSIM_BUILD_CONSOLE)
    file(GLOB MARKETSIM_UI_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/UI/*.cpp")
    add_executable(marketsim main.cpp ${MARKETSIM_UI_SOURCES})
    target_link_libraries(marketsim PRIVATE marketsim_core)
    target_compile_options(marketsim PRIVATE ${MARKETSIM_WARNINGS})
endif()

# ---------------------------------------------------------------------------
# Qt 6 GUI. Skipped (with a warning) if Qt 6 is not installed, so the console
# build never depends on Qt.
# ---------------------------------------------------------------------------
if(MARKETSIM_BUILD_GUI)
    find_package(Qt6 6.2 QUIET COMPONENTS Widgets)
    if(Qt6_FOUND)
        set(CMAKE_AUTOMOC ON)
        file(GLOB MARKETSIM_GUI_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/GUI/*.cpp")
        file(GLOB MARKETSIM_GUI_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/GUI/*.h")
        qt_add_executable(marketsim_gui ${MARKETSIM_GUI_SOURCES} ${MARKETSIM_GUI_HEADERS})
        target_link_libraries(marketsim_gui PRIVATE marketsim_core Qt6::Widgets)
        target_compile_options(marketsim_gui PRIVATE ${MARKETSIM_WARNINGS})
    else()
        message(WARNING "Qt 6 (Widgets) not found - skipping marketsim_gui. "
                        "Set CMAKE_PREFIX_PATH to your Qt 6 install to build the GUI.")
    endif()
endif()
